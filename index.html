2. app/src/main/java/pk/zong/signalmonitor/MainActivity.kt (Live 3G/4G Tower Reader) 
package pk.zong.signalmonitor

import android.Manifest
import android.annotation.SuppressLint
import android.content.Context
import android.content.pm.PackageManager
import android.os.Build
import android.os.Bundle
import android.telephony.*
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import androidx.core.app.ActivityCompat

class MainActivity : AppCompatActivity() {

    private lateinit var telephonyManager: TelephonyManager
    private lateinit var statusTextView: TextView
    private var previousDbm: Int? = null

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        statusTextView = TextView(this).apply {
            textSize = 16f
            setPadding(48, 64, 48, 64)
        }
        setContentView(statusTextView)

        telephonyManager = getSystemService(Context.TELEPHONY_SERVICE) as TelephonyManager
        checkPermissionsAndStart()
    }

    private fun checkPermissionsAndStart() {
        val perms = arrayOf(
            Manifest.permission.READ_PHONE_STATE,
            Manifest.permission.ACCESS_FINE_LOCATION
        )
        if (perms.any { ActivityCompat.checkSelfPermission(this, it) != PackageManager.PERMISSION_GRANTED }) {
            ActivityCompat.requestPermissions(this, perms, 101)
        } else {
            readZongTowerSignal()
        }
    }

    // NOTE: In the full Android Studio ZIP package, MainActivity loads this exact 4-SIM HTML UI
    // inside Android WebView (file:///android_asset/index.html) and bridges live TelephonyManager
    // RSRP, RSRQ, SINR & RSCP readings for Zong (410-04), Jazz (410-01), Telenor (410-06) & Ufone (410-03)
    // directly into the 4 separate meters via window.updateFromAndroidHardware(jsonString)!
    @SuppressLint("MissingPermission")
    private fun readZongTowerSignal() {
        val cellInfoList: List<CellInfo> = telephonyManager.allCellInfo ?: emptyList()
        val report = StringBuilder("📡 Pakistan All-SIM (Zong/Jazz/Telenor/Ufone) Signal Monitor\n\n")

        for (cell in cellInfoList) {
            if (!cell.isRegistered) continue

            when (cell) {
                is CellInfoLte -> {
                    val id: CellIdentityLte = cell.cellIdentity
                    val strength: CellSignalStrengthLte = cell.cellSignalStrength
                    val mcc = if (Build.VERSION.SDK_INT >= 28) id.mccString else id.mcc.toString()
                    val mnc = if (Build.VERSION.SDK_INT >= 28) id.mncString else id.mnc.toString()

                    val rsrp = strength.rsrp // -140 to -43 dBm
                    val rsrq = strength.rsrq // -20 to -3 dB
                    val sinr = strength.rssnr // -20 to +30 dB
                    val bars = strength.level // 0..4 bars
                    val delta = previousDbm?.let { rsrp - it } ?: 0
                    previousDbm = rsrp

                    val carrierName = when (mnc) {
                        "01", "07", "1", "7" -> "Jazz 4G (PMCL)"
                        "03", "3" -> "Ufone 4G (PTML)"
                        "04", "4" -> "Zong 4G (CMPak)"
                        "06", "6" -> "Telenor Pakistan"
                        else -> "Pakistan Operator ($mcc-$mnc)"
                    }
                    report.append("SIM Network: $carrierName (PLMN $mcc-$mnc)\n")
                    report.append("Cell ID (ECI): ${id.ci} | Earfcn: ${id.earfcn}\n")
                    report.append("Signal Strength (RSRP): $rsrp dBm ($bars/4 Bars)\n")
                    report.append("Signal Quality (RSRQ): $rsrq dB | SINR: $sinr dB\n")
                    report.append("Variation Delta: ${if (delta >= 0) "+$delta" else "$delta"} dB\n")
                    // Check if network will run fast or slow even with full signal:
                    val speedNote = when {
                        sinr >= 13 && rsrq >= -12 -> "FAST (Tez Chalega - Low Tower Load)"
                        sinr in 3..12 || rsrq in -15..-13 -> "MEDIUM (Guzaara Chalega)"
                        else -> "SLOW (Signal theek hai magar Tower Rush/Interference ki wajah se slow chalega)"
                    }
                    report.append("Speed Prediction: $speedNote\n")
                }
                is CellInfoWcdma -> {
                    val id: CellIdentityWcdma = cell.cellIdentity
                    val strength: CellSignalStrengthWcdma = cell.cellSignalStrength
                    val rscp = strength.dbm // -120 to -24 dBm
                    val bars = strength.level // 0..4 bars
                    val delta = previousDbm?.let { rscp - it } ?: 0
                    previousDbm = rscp

                    report.append("Network: Zong 3G UMTS/HSPA+ (Band 1 - 2100 MHz)\n")
                    report.append("Cell ID: ${id.cid} | Uarfcn: ${id.uarfcn}\n")
                    report.append("Signal Strength (RSCP): $rscp dBm ($bars/4 Bars)\n")
                    report.append("Variation Delta: ${if (delta >= 0) "+$delta" else "$delta"} dB\n")
                }
            }
        }
        statusTextView.text = report.toString()
    }
}
